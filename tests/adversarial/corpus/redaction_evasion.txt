# Every shape that reached a redacted column during adversarial review, and the ones that never did.
# Run with MCP_REDACT_COLUMNS set (the runner does this). Column name assumed: {{COL}}.
#
# expect	SQL	why this case exists
deny-under-redaction	SELECT {{COL}} FROM {{T}} LIMIT 1	the direct reference
deny-under-redaction	SELECT "PASSWORD" FROM {{T}} LIMIT 1	quoting and case must not matter
deny-under-redaction	SELECT {{COL}} AS pw FROM {{T}} LIMIT 1	renaming defeats a filter that inspects result keys
deny-under-redaction	SELECT md5({{COL}}) FROM {{T}} LIMIT 1	wrapping in a function
deny-under-redaction	SELECT left({{COL}}, 4) FROM {{T}} LIMIT 1	a prefix is still the value, one character at a time
deny-under-redaction	SELECT 1 FROM {{T}} WHERE {{COL}} = 'x'	an oracle: the value never leaves, the answer does
deny-under-redaction	SELECT 1 FROM {{T}} ORDER BY {{COL}}	ordering leaks the same way
deny-under-redaction	SELECT row_to_json(t) FROM {{T}} t LIMIT 1	round 6: the whole row as one nested value
deny-under-redaction	SELECT to_jsonb(t) FROM {{T}} t LIMIT 1	same, other function
deny-under-redaction	SELECT json_agg(t) FROM {{T}} t	same, and it returns every row at once
deny-under-redaction	SELECT t::text FROM {{T}} t LIMIT 1	round 7: cast the record to text and the tree becomes one leaf
deny-under-redaction	SELECT {{T}}::text FROM {{T}} LIMIT 1	same, using the table name rather than an alias
deny-under-redaction	SELECT row_to_json(t)::text FROM {{T}} t LIMIT 1	JSON first, then flattened
deny-under-redaction	SELECT json_agg(t)::text FROM {{T}} t	the whole column in one string
deny-under-redaction	SELECT to_jsonb(t) -> '{{COL}}' FROM {{T}} t LIMIT 1	round 7: the name as a literal, not an identifier
deny-under-redaction	SELECT to_jsonb(t) ->> '{{COL}}' AS x FROM {{T}} t LIMIT 1	same, text extraction
deny-under-redaction	SELECT to_jsonb(t) #>> '{{{COL}}}' AS x FROM {{T}} t LIMIT 1	path syntax names the key differently
deny-under-redaction	SELECT jsonb_path_query(to_jsonb(t), '$.{{COL}}') FROM {{T}} t LIMIT 1	jsonpath names it a third way
deny-under-redaction	SELECT string_agg(to_jsonb(t) ->> '{{COL}}', ',') FROM {{T}} t	aggregation over the same trick
deny-under-redaction	SELECT array_agg(to_jsonb(t) ->> '{{COL}}') FROM {{T}} t	and again as an array
deny-under-redaction	SELECT json_build_object('safe', to_jsonb(t) -> '{{COL}}') FROM {{T}} t LIMIT 1	hidden under an innocent key
deny-under-redaction	SELECT md5(to_jsonb(t) ->> '{{COL}}') FROM {{T}} t LIMIT 1	a digest is still a channel
deny-under-redaction	WITH x AS (SELECT to_jsonb(t) AS j FROM {{T}} t) SELECT j ->> '{{COL}}' FROM x	one level of indirection
deny-under-redaction	SELECT ROW(t.*)::text FROM {{T}} t LIMIT 1	round 8: a qualified wildcard is a different AST node than an identifier
deny-under-redaction	SELECT to_jsonb(t.*)::text FROM {{T}} t LIMIT 1	same shape, other function
deny-under-redaction	SELECT row_to_json({{T}}.*)::text FROM {{T}} LIMIT 1	same shape, table-qualified
deny-under-redaction	SELECT x.c9 FROM (SELECT * FROM {{T}}) AS x(c1,c2,c3,c4,c5,c6,c7,c8,c9,c10,c11) LIMIT 1	positional renaming: the name never appears anywhere
deny-under-redaction	WITH x(c1,c2,c3,c4,c5,c6,c7,c8,c9,c10,c11) AS (SELECT * FROM {{T}}) SELECT c9 FROM x LIMIT 1	the same, through a CTE
allow	SELECT 1 AS x FROM {{T}} LIMIT 1	an ordinary query must keep working
allow	SELECT count(*) FROM {{T}}	so must an aggregate
allow	SELECT * FROM {{T}} LIMIT 1	the value is masked here, not refused: the column is never named
deny	SELECT get_raw_page('{{T}}', 0)	round 0.1.7: 8192 bytes of the page, with the redacted column in plain ASCII — demonstrated, not theorised
deny	SELECT * FROM heap_page_items(get_raw_page('{{T}}', 0))	the same data, one layer up
deny	SELECT tuple_data_split('{{T}}'::regclass, t_data, t_infomask, t_infomask2, t_bits) FROM heap_page_items(get_raw_page('{{T}}', 0))	and split back into fields
deny	SELECT page_header(get_raw_page('{{T}}', 0))	.
deny	SELECT most_common_vals FROM pg_stats WHERE tablename='{{T}}'	round 0.1.7: planner statistics hold REAL values sampled from the column
deny	SELECT histogram_bounds FROM pg_stats WHERE tablename='{{T}}'	.
deny	SELECT stavalues1 FROM pg_statistic	the same, one layer below the view
deny	SELECT chunk_data FROM pg_toast.pg_toast_16384	round 0.1.7: a value too long for its row lives in TOAST, where columns no longer exist
deny	SELECT * FROM pg_toast.pg_toast_16384	.
deny	SELECT data FROM pg_largeobject	the bytes underneath lo_get
deny	SELECT stxdmcv FROM pg_statistic_ext_data	extended statistics keep their own sample of real values
deny	SELECT count(*) FROM {{T}} JOIN (SELECT 'probe' AS {{COL}}) g USING ({{COL}})	round 0.1.8: an equality oracle needing NO privileges, unlike the reads that go underneath columns
deny	SELECT count(*) FROM {{T}} LEFT JOIN q USING ({{COL}})	.
deny	SELECT count(*) FROM {{T}} NATURAL JOIN q	the schema decides what a natural join joins on, and this validator has no schema
