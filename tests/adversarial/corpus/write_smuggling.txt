# Attempts to change data through a server that only reads.
#
# expect	SQL	why this case exists
deny	INSERT INTO {{T2}} VALUES ('x')	the obvious one
deny	UPDATE {{T2}} SET x = 1	and its siblings
deny	DELETE FROM {{T2}}	.
deny	DROP TABLE {{T2}}	.
deny	TRUNCATE {{T2}}	.
deny	CREATE TABLE t (x int)	.
deny	ALTER TABLE {{T2}} ADD COLUMN zz int	.
deny	GRANT SELECT ON {{T2}} TO PUBLIC	privilege changes are writes too
deny	WITH d AS (DELETE FROM {{T2}} RETURNING *) SELECT * FROM d	a data-modifying CTE parses as a query
deny	WITH u AS (UPDATE {{T2}} SET x=1 RETURNING *) SELECT * FROM u	.
deny	SELECT 1; DROP TABLE {{T2}}	two statements in one message
deny	SELECT * FROM {{T2}} FOR UPDATE	a row lock is a write intent
deny	EXPLAIN ANALYZE INSERT INTO {{T2}} VALUES ('x')	ANALYZE executes what it explains
deny	COMMIT	transaction control ends the read-only transaction we rely on
deny	COMMIT WORK	.
deny	COMMIT TRANSACTION	.
deny	COMMIT AND CHAIN	.
deny	END	.
deny	END WORK	.
deny	END TRANSACTION	.
deny	END AND NO CHAIN	.
deny	ROLLBACK	.
deny	ABORT	.
allow	SELECT 1 AS x FROM {{T2}} LIMIT 1	reading still works
allow	WITH x AS (SELECT 1 AS x FROM {{T2}} LIMIT 1) SELECT * FROM x	so does a read-only CTE
deny	SELECT cron.schedule('nightly', '0 0 * * *', 'DROP TABLE users')	round 0.1.7: pg_cron lives in its own schema, so its names matched no pg_* family and this was ALLOWED
deny	SELECT cron.schedule_in_database('j', '0 0 * * *', 'DELETE FROM t', 'postgres')	the same, aimed at another database
deny	SELECT cron.unschedule('nightly')	removing somebody else's job is a write too
deny	SELECT cron.alter_job(1, schedule := '0 0 * * *')	.
deny	SELECT CRON.SCHEDULE('a', 'b', 'c')	the schema match must not care about case
deny	SELECT * FROM repack.repack_table('t')	pg_repack rewrites the table in place
deny	SELECT drop_chunks('{{T2}}', older_than => now())	TimescaleDB installs into public, so a data-deleting call looks like any other
deny	SELECT create_hypertable('{{T2}}', 'time')	.
deny	SELECT compress_chunk('_hyper_1_1_chunk')	.
deny	SELECT add_retention_policy('{{T2}}', INTERVAL '7 days')	a policy that deletes data on a schedule
deny	SELECT AddGeometryColumn('{{T2}}', 'geom', 4326, 'POINT', 2)	PostGIS wrapper over ALTER TABLE
deny	SELECT DropGeometryColumn('{{T2}}', 'geom')	.
deny	SELECT partman.run_maintenance()	creates and drops partitions
deny	SELECT pglogical.create_subscription('s', 'host=evil')	replication to somewhere else entirely
deny	SELECT squeeze.squeeze_table('public', '{{T2}}')	rewrites the table in place
deny	SELECT repeat('x', 100000000)	round 0.1.8: folded during planning, 95 MB the server could never return
deny	SELECT repeat('x', 10000 * 10000)	arithmetic walks past a check on the literal alone
deny	SELECT repeat(repeat(repeat('x',1000),1000),1000)	49 bytes of SQL, 1 GB plan, 5.9 GB of backend memory measured
deny	SELECT lpad('x', 500000000)	repeat is not the only expansion function
deny	SELECT repeat('ab', 5000000) || repeat('cd', 5000000)	concatenation adds
allow	SELECT repeat('-', 80)	a separator is not an attack, and the rule must not cost this
